	CJK_RE = re.compile(r"[\u3040-\u30ff\u3400-\u4dbf\u4e00-\u9fff\uac00-\ud7af\uff00-\uffef]+")
	JSON_KEYS = "fact|numbers_dates|year|type|sources|evidence|conflict"
	YEAR_RUN_RE = re.compile(r"(?:\b\d{3,4}\b\s*,\s*){6,}\b\d{3,4}\b")   # >= 7 số 3-4 chữ số liên tiếp: danh sách năm bị lặp

	def strip_echo(text):
		# bỏ phần suy nghĩ của model reasoning
		text = re.sub(r"<think>.*?</think>", "", text, flags=re.S)
		# xóa dòng JSON mẫu bị chép lại (fact = "...")
		text = re.sub(r'\[\s*\{\s*"fact"\s*:\s*"(?:\.\.\.|…)"[^\n]*\]', "", text)
		# bỏ mọi thứ nằm trước ###SECTION đầu tiên
		m = re.search(r"#{3}\s*SECTION", text, flags=re.I)
		return text[m.start():] if m else text

	def sanitize_text(text):
		"""Dọn text của 1 section: JSON FACTS lọt vào, chuỗi số lặp, chữ Hán/Hàn, nhãn cấu trúc, mã nguồn S25."""
		text = str(text)
		# 1) chuỗi năm/số dài (LLM đếm vô hạn)
		text = YEAR_RUN_RE.sub("", text)
		# 2) object JSON {"fact": ...} còn sót: xóa tới hết dòng
		text = re.sub(r'\[?\s*\{\s*"fact"[^\n]*', "", text)
		# 3) mảnh key/value JSON còn sót sau khi xóa danh sách (vd: ], "type": "data", "sources": [...]})
		text = re.sub(
			r'\]?\s*,?\s*"(?:' + JSON_KEYS + r')"\s*:\s*(?:"[^"]*"|\[[^\]]*\]|-?\d+|null|true|false)\s*,?\s*[\]\}]*',
			" ", text)
		# 4) ký tự Hán/Hàn/Nhật (code-switching của model đa ngữ)
		text = CJK_RE.sub("", text)
		# 5) nhãn cấu trúc từ prompt bị in ra, dấu ba chấm đầu đoạn
		text = re.sub(r"(?im)^\s*(?:câu hỏi bỏ lửng|góc nhìn đối lập|nghịch lý|kết nối|chuyển đoạn)\s*:\s*", "", text)
		text = re.sub(r"(?m)^\s*(?:\.{3,}|…)\s*", "", text)
		# 6) nhắc tên nguồn trần trụi trong lời văn ("số liệu từ S25")
		text = re.sub(r"\b(?:từ|theo|của)\s+S\d{1,3}\b", "", text)
		# 7) dọn khoảng trắng nhưng giữ xuống dòng
		text = re.sub(r"[ \t]{2,}", " ", text)
		text = re.sub(r"[ \t]+([,.;:!?])", r"\1", text)
		text = re.sub(r"\n{3,}", "\n\n", text)
		return text.strip()

	def is_degenerate(text):
		"""True nếu text có dấu hiệu hỏng: còn JSON, quá nhiều số, hoặc lặp cụm từ."""
		if not text:
			return False
		if re.search(r'"(?:fact|numbers_dates|evidence|sources)"\s*:', text):
			return True
		if YEAR_RUN_RE.search(text):
			return True
		toks = text.split()
		if len(toks) >= 30:
			num_toks = sum(1 for t in toks if re.fullmatch(r"[\d.,/%\-–]+", t))
			if num_toks / len(toks) > 0.3:
				return True
			tri = Counter(zip(toks, toks[1:], toks[2:]))
			if tri and tri.most_common(1)[0][1] >= 8:
				return True
		return False

	def sanitize_facts(facts):
		"""Chuẩn hóa FACTS: cắt numbers_dates, kiểm tra năm hợp lệ, bỏ fact hỏng."""
		out = []
		for f in facts:
			if not isinstance(f, dict) or not f.get("fact"):
				continue
			fact_txt = str(f["fact"])
			if YEAR_RUN_RE.search(fact_txt):
				continue   # fact chứa danh sách năm bịa: bỏ
			f["fact"] = fact_txt[:500]
			nd = f.get("numbers_dates", "")
			if isinstance(nd, list):
				nd = ", ".join(str(x) for x in nd[:5])
			nd = str(nd or "")
			nums = re.findall(r"\d[\d.,/]*", nd)
			if len(nums) > 6 or len(nd) > 200:
				nd = ", ".join(nums[:5])
			f["numbers_dates"] = nd
			try:
				y = int(f.get("year")) if f.get("year") is not None else None
			except Exception:
				y = None
			if y is not None and not (1900 <= y <= current_year + 1):
				y = None
			f["year"] = y
			f["evidence"] = " ".join(str(f.get("evidence", "")).split()[:25])
			out.append(f)
		return out

	# lấy text an toàn: .text có thể là property hoặc method, content có thể là list (Gemini)
	def llm_call(prompt, label=""):
		st.write(f"  [LLM:{label}] prompt = {len(prompt):,} ký tự (~{len(prompt) // 3:,} token)")
		response = llm.invoke(prompt)
		text = getattr(response, "text", None)
		if callable(text):
			text = text()
		if not text:
			content_ = getattr(response, "content", "")
			if isinstance(content_, list):
				text = "".join(
					c.get("text", "") if isinstance(c, dict) else str(c) for c in content_
				)
			else:
				text = content_
		return strip_echo(str(text or ""))

	def clean_for_tts(text):
		text = sanitize_text(text)   # [SỬA4] lọc rác lần cuối trước khi đưa vào TTS
		text = re.sub(r"\s*\[S\d+(?:\s*,\s*S\d+)*\]", "", text)
		text = re.sub(r"\b[Ss]\d{1,3}\b", "", text)   # [SỬA4] S25 trần trụi
		text = re.sub(r"https?://\S+", "", text)
		text = re.sub(r"^#{1,6}\s*", "", text, flags=re.M)
		text = re.sub(r"[*_`>]+", "", text)
		text = re.sub(r"^\s*[-•]\s+", "", text, flags=re.M)
		text = re.sub(r"[{}\[\]]", "", text)   # [SỬA4] ngoặc JSON còn sót
		lines = [ln.strip() for ln in text.splitlines() if ln.strip()]
		lines = [ln if re.search(r"[.!?…:;]$", ln) else ln + "." for ln in lines]
		text = " ".join(lines)
		return re.sub(r"\s{2,}", " ", text).strip()

	def tokenize(text):
		toks = re.findall(r"\w+", str(text).lower())
		return toks + [a + "_" + b for a, b in zip(toks, toks[1:])]

	def build_index(docs):
		splitter = RecursiveCharacterTextSplitter(chunk_size=1200, chunk_overlap=100)
		chunks = []
		for item in docs:
			for chunk_index, chunk in enumerate(splitter.split_text(item["markdown"])):
				if len(chunk.strip()) < 150:
					continue
				toks = tokenize(chunk)
				chunks.append({
					"text": chunk,
					"source": item["url"],
					"title": item["title"],
					"chunk_id": f"{item['url']}#{chunk_index}",
					"tf": Counter(toks),
					"len": max(len(toks), 1),
				})
		st.write("THE TOTAL NUMBER OF CONTENT MARKDOWN CHUNKS:", len(chunks))
		if not chunks:
			raise ValueError("Không tìm thấy dữ liệu web để tạo RAG.")
		df = Counter()
		for c in chunks:
			df.update(c["tf"].keys())
		return {
			"chunks": chunks, "df": df, "N": len(chunks),
			"avgdl": sum(c["len"] for c in chunks) / len(chunks),
		}

	def bm25_scores(index, query, k1=1.5, b=0.75):
		q_tokens = set(tokenize(query))
		N, df, avgdl = index["N"], index["df"], index["avgdl"]
		scores = []
		for c in index["chunks"]:
			s = 0.0
			for t in q_tokens:
				f = c["tf"].get(t, 0)
				if not f:
					continue
				n_t = df.get(t, 0)
				idf = math.log(1 + (N - n_t + 0.5) / (n_t + 0.5))
				s += idf * f * (k1 + 1) / (f + k1 * (1 - b + b * c["len"] / avgdl))
			scores.append(s)
		return scores

	emb_cache = {}
	def rerank_with_embeddings(query, cands):
		import numpy as np
		if "model" not in emb_cache:
			emb_cache["model"] = langchain_embeddings_jina_ai()
		emb = emb_cache["model"]
		qv = np.array(emb.embed_query(query))
		dv = np.array(emb.embed_documents([c["text"] for c in cands]))
		sims = dv @ qv / (np.linalg.norm(dv, axis=1) * np.linalg.norm(qv) + 1e-9)
		return [cands[i] for i in np.argsort(-sims)]

	global_source_ids = {}
	source_text = {}
	source_seen = set()
	def get_section_context(index, query, max_chunks_per_source=2, k=5, pool=30, max_chars=None, skip_seen=False):
		max_chars = max_chars or max_chars_per_section
		query = str(query)[:300]
		scores = bm25_scores(index, query)
		ranked = sorted(range(len(scores)), key=lambda i: scores[i], reverse=True)
		cands = [index["chunks"][i] for i in ranked[:pool] if scores[i] > 0]
		if not cands:
			return ""
		if use_real_embeddings:
			try:
				cands = rerank_with_embeddings(query, cands)
			except Exception as e:
				st.write(f"Rerank embedding lỗi, dùng thứ hạng BM25: {e}")
		count, parts, total = {}, [], 0
		for c in cands:
			if skip_seen and c["chunk_id"] in source_seen:
				continue
			src = c["source"]
			count[src] = count.get(src, 0) + 1
			if count[src] > max_chunks_per_source:
				continue
			if src not in global_source_ids:
				global_source_ids[src] = f"S{len(global_source_ids) + 1}"
			sid = global_source_ids[src]
			part = f"[{sid}] ({c['title']}) {src}\n{c['text']}"
			if parts and total + len(part) > max_chars:
				break
			if c["chunk_id"] not in source_seen:
				source_seen.add(c["chunk_id"])
				source_text[sid] = source_text.get(sid, "") + "\n" + c["text"]
			parts.append(part)
			total += len(part)
			if len(parts) >= k:
				break
		return "\n\n".join(parts)

	def tag_freshness(facts):
		for f in facts:
			y = None
			try:
				y = int(f.get("year")) if f.get("year") is not None else None
			except Exception:
				y = None
			if y is None:
				years = re.findall(r"\b(?:19|20)\d{2}\b", str(f.get("numbers_dates", "")))
				years = [int(v) for v in years if int(v) <= current_year + 1]   # [SỬA4] bỏ năm phi lý
				y = max(years) if years else None
			if y is None:
				f["freshness"] = "evergreen" if f.get("type") == "concept" else "unknown"
			elif y >= recent_from:
				f["freshness"] = "current"
			else:
				f["freshness"] = "historical"
			f["year"] = y
		return facts

	# [SỬA4] tách JSON FACTS bị lọt vào text (bằng bracket-matching), phần text còn lại vẫn giữ
	def split_text_and_facts(body):
		m = re.search(r'\[?\s*\{\s*"fact"\s*:', body)
		if not m:
			return body, ""
		start = m.start()
		depth, in_str, esc, end = 0, False, False, len(body)
		for i in range(start, len(body)):
			ch = body[i]
			if in_str:
				if esc:
					esc = False
				elif ch == "\\":
					esc = True
				elif ch == '"':
					in_str = False
				continue
			if ch == '"':
				in_str = True
			elif ch in "[{":
				depth += 1
			elif ch in "]}":
				depth -= 1
				if depth <= 0:
					end = i + 1
					break
		return body[:start] + "\n" + body[end:], body[start:end]

	def load_facts(body):
		try:
			facts = parse_json_loose(body)
		except Exception:
			facts = []
		if isinstance(facts, dict):
			facts = [facts]
		if not isinstance(facts, list):
			return []
		return [f for f in facts if isinstance(f, dict) and f.get("fact")]

	# [SỬA4] parse_batch chịu lỗi: marker dính cùng dòng, JSON lọt vào text, text hỏng bị loại để retry
	def parse_batch(raw):
		out = {}
		raw = re.sub(r"```\w*", "", str(raw))
		raw = raw.replace("**", "")
		# đưa mọi marker xuống dòng riêng (model hay in marker dính vào câu trước/sau)
		raw = re.sub(r"[ \t]*(#{3}\s*(?:SECTION|FACTS)\s*\|\s*\w+)[ \t]*", r"\n\1\n", raw, flags=re.I)
		parts = re.split(r"^\s*#{3}\s*(SECTION|FACTS)\s*\|\s*(\w+)\s*$", raw, flags=re.M | re.I)
		for j in range(1, len(parts) - 2, 3):
			kind, sid, body = parts[j].upper(), parts[j + 1], parts[j + 2]
			slot = out.setdefault(sid, {"text": "", "facts": []})
			if kind == "SECTION":
				text_part, leaked = split_text_and_facts(body)
				if not slot["text"]:
					slot["text"] = text_part.strip()
				if leaked:
					slot["facts"] += load_facts(leaked)
			else:
				slot["facts"] += load_facts(body)

		for sid, slot in out.items():
			clean = sanitize_text(slot["text"])
			if clean and is_degenerate(clean):
				st.write(f"  ⚠ Section {sid}: text hỏng (lặp số/JSON/lặp cụm), loại để viết lại")
				clean = ""
			slot["text"] = clean
			slot["facts"] = sanitize_facts(slot["facts"])

		if not out and raw.strip():
			st.write("  ⚠ parse_batch không tách được section nào. Raw (500 ký tự đầu):")
			st.write(raw[:500])
		return out

	num_cache, norm_cache = {}, {}
	def get_source_nums(sid):
		if sid not in num_cache:

	def find_unsupported_numbers(text):
		"""Trả về các câu có số/ngày KHÔNG xuất hiện trong nguồn mà đoạn đó trích dẫn."""
		bad, all_nums = [], None
		for para in re.split(r"\n+", text):
			cited = set()
			for grp in re.findall(r"\[(S\d+(?:\s*,\s*S\d+)*)\]", para):
				cited.update(re.findall(r"S\d+", grp))
			cited = {s for s in cited if s in source_text}
			if cited:
				hay = set().union(*(get_source_nums(s) for s in cited))
			else:
				if all_nums is None:
					all_nums = set().union(*(get_source_nums(s) for s in source_text)) if source_text else set()
				hay = all_nums
			for sent in re.split(r"(?<=[.!?])\s+", para):
				low = sent.lower().strip()
				# [SỬA4] chỉ miễn kiểm tra khi câu THỰC SỰ mở bằng "giả sử"/"ví dụ" (tránh LLM lách bằng cách chèn từ này giữa câu)
				if low.startswith(("giả sử", "ví dụ", "chẳng hạn")) or "giả sử" in low[:40]:
					continue
				clean = re.sub(r"\[S[^\]]*\]", "", sent)
				nums = {re.sub(r"\D", "", x) for x in re.findall(r"\d[\d.,]*", clean)}
				nums = {n for n in nums if len(n) >= 2}
				if nums - hay:
					bad.append(sent.strip())
		return bad

	def check_section(text, facts):
		issues = find_unsupported_numbers(text)
		for f in facts:
			ev = norm(f.get("evidence", ""))
			srcs = f.get("sources") or []
			if isinstance(srcs, str):
				srcs = [srcs]
			srcs = [s for s in srcs if s in source_text]
			f["evidence_ok"] = bool(ev) and any(ev in get_norm_source(s) for s in srcs)
		bad_ev = sum(1 for f in facts if not f["evidence_ok"])
		if facts and bad_ev > len(facts) / 2:
			issues.append(f"{bad_ev}/{len(facts)} dữ kiện không có evidence khớp nguồn")
		return issues, [f for f in facts if f["evidence_ok"]]

4. Nếu số liệu của năm cũ bị gọi là "hiện nay"/"năm nay", sửa lại cho nêu rõ năm.
5. Chỉ xóa/sửa các câu thật sự sai hoặc không có trong nguồn. KHÔNG rút gọn, KHÔNG tóm tắt các câu còn lại.
6. Câu nào mang giọng chỉ trích, phán xét, quy trách nhiệm cá nhân/tổ chức/nhà nước thì viết lại thành mô tả trung tính, giữ nguyên dữ kiện và số liệu.
7. Xóa mọi chuỗi liệt kê năm/số dài vô nghĩa, mọi mảnh JSON, và mọi ký tự Hán/Hàn/Nhật nếu có.

Chỉ trả JSON:
{{"unsupported_claims": ["..."], "clean_text": "..."}}"""
		raw = llm_call(prompt, label="verify")
		if not raw:
			return draft, ["verify_failed_empty"], False
		try:
			data = parse_json(raw)
			new_text = sanitize_text(data.get("clean_text") or draft)   # [SỬA4] sanitize cả kết quả verify
			return new_text, data.get("unsupported_claims", []), True
		except Exception:
			return draft, ["verify_parse_error"], False

	# =========================================================
	# 7.PIPELINE
	# =========================================================
	# 7.2 Search + index
	docs = search_web(list_keywords, max_results=web_search_max_results)
	if not docs:
		return fail("Không có tài liệu web nào để tạo RAG.")
	try:
		index = build_index(docs)
	except ValueError as e:
		return fail(str(e))

	# 7.3 Truy xuất context cho từng section
	n_est = len(list_keywords) + (1 if add_advice_section else 0)
	if target_total_words:
		target_words_per_outline = max(300, int(target_total_words) // max(n_est, 1))
		ctx_chars = max(max_chars_per_section, int(target_words_per_outline * chars_per_word))
		k_ctx = ctx_chars // 1100 + 1
		pool_ctx = max(30, k_ctx * 6)
		max_chars_per_call = max(max_chars_per_call, ctx_chars * max_sections_per_call + 1000)
		max_words_per_call = max(max_words_per_call, target_words_per_outline * max_sections_per_call + 100)
		st.write(f"MỤC TIÊU: {target_words_per_outline} từ/section, context {ctx_chars:,} ký tự, {k_ctx} chunk")
	else:
		ctx_chars, k_ctx, pool_ctx = max_chars_per_section, 5, 30

	planned, skipped = [], []
	for i, query in enumerate(list_keywords):
		st.write(f"[{i + 1}/{len(list_keywords)}] RETRIEVE: {query}")
		context = get_section_context(index, query, k=k_ctx, pool=pool_ctx, max_chars=ctx_chars, skip_seen=True)
		if not context:
			context = get_section_context(index, query, k=k_ctx, pool=pool_ctx, max_chars=ctx_chars)
		title = f"Góc nhìn về: {query}"
		if not context:
			st.write("  → Không có chunk liên quan, bỏ qua section này")
			skipped.append({"query": query, "title": title, "text": "", "words": 0,
							"removed_claims": [], "status": "insufficient_data", "reason": "no_chunks"})
			continue
		planned.append({"kind": "section", "query": query, "title": title, "context": context})

	if not planned:
		return fail("Không section nào đủ dữ kiện.")

	total = len(planned)
	items = []
	for i, p in enumerate(planned):
		p["id"] = str(i + 1)
		p["target"] = target_words_per_outline
		p["flow"] = get_content_flow(i, total)
		p["next_topic"] = (planned[i + 1]["query"] if i + 1 < total
						   else (ADVICE_TITLE if add_advice_section else ""))
		items.append(p)

	if add_advice_section:
		advice_context = get_section_context(index, f"{str(raw_topic)[:300]} xu hướng rủi ro số liệu mới nhất",
											 k=k_ctx, pool=pool_ctx, max_chars=ctx_chars)
		if not advice_context:
			advice_context = "\n\n".join(p["context"][:1000] for p in planned)[:ctx_chars]
		items.append({"kind": "advice", "id": "ADVICE", "query": "advice", "title": ADVICE_TITLE,
					  "context": advice_context, "target": target_words_per_outline,
					  "flow": "", "next_topic": ""})
	for n, it in enumerate(items):
		it["pos"] = n
	outline = [it["title"] for it in items]

	# 7.4 Viết: gom section thành batch
	def make_batches(its):
		out, cur, chars, words = [], [], 0, 0
		for it in its:
			c = len(it["context"])
			if cur and (
				chars + c > max_chars_per_call
				or words + it["target"] > max_words_per_call
				or len(cur) >= max_sections_per_call
			):
				out.append(cur)
				cur, chars, words = [], 0, 0
			cur.append(it)
			chars += c
			words += it["target"]
		if cur:
			out.append(cur)
		return out

	batches = make_batches(items)
	st.write(f"WRITE: {len(items)} section trong {len(batches)} call (tối đa {max_workers} luồng)")

	# [SỬA4] write_errors dùng chung giữa các thread -> thêm lock
	write_errors = []
	err_lock = threading.Lock()
	def add_err(msg):
		with err_lock:
			write_errors.append(msg)

	def safe_write(batch):
		ids = [b["id"] for b in batch]
		try:
			raw = write_batch(batch, outline)
			if raw and raw.strip():
				return raw
			add_err(f"Batch {ids} trả về rỗng")
			return ""
		except Exception as e:
			if len(batch) > 1:
				mid = len(batch) // 2
				add_err(f"Batch {ids} lỗi ({type(e).__name__}: {str(e)[:150]}), chia đôi thử lại")
				return safe_write(batch[:mid]) + "\n" + safe_write(batch[mid:])
			add_err(f"Batch {ids} lỗi: {type(e).__name__}: {e}")
			return ""

	def run_write(batch_list):
		with ThreadPoolExecutor(max_workers=max(1, min(max_workers, len(batch_list)))) as ex:
			raws_ = list(ex.map(safe_write, batch_list))
		for err in write_errors:
			st.write(f"  ⚠ {err}")
		write_errors.clear()
		out = {}
		for raw in raws_:
			out.update(parse_batch(raw))
		return out

	parsed = run_write(batches)

	# section thiếu / text rỗng (kể cả text bị loại vì hỏng) được viết lại, tối đa 2 lần
	for _retry in range(2):
		missing = [it for it in items if it["id"] not in parsed or not parsed[it["id"]]["text"].strip()]
		if not missing:
			break
		st.write(f"  → Thiếu/rỗng/hỏng {len(missing)} section trong kết quả, viết lại (lần {_retry + 1})")
		parsed.update(run_write(make_batches(missing)))

	# 7.4b MỞ RỘNG
	def expand_prompt(p, prev_text, ctx, need):
		return f"""Bạn là {llm_role}, đồng thời là người viết kịch bản đọc thành tiếng. Hôm nay là {today_str}.
CHỦ ĐỀ CHÍNH CỦA VIDEO: {topic_short}
Ngôn ngữ: {language_name}. Văn phong kể chuyện, dễ đọc thành tiếng.

Section "{p['title']}" đã có phần viết sau (đoạn cuối):
--- PHẦN ĐÃ VIẾT ---
{prev_text[-1200:]}
--- HẾT ---

Hãy viết THÊM khoảng {need} từ cho section này. Đoạn mới sẽ được chèn TRƯỚC câu kết của section, nên: không chào, không kết bài, không CTA, không tổng kết, không lời khuyên hành động.
CHỈ dùng dữ kiện trong DỮ LIỆU MỚI bên dưới và KHÔNG lặp lại dữ kiện đã nêu ở phần đã viết.
Nếu dữ liệu mới có ít hơn {min_facts} dữ kiện liên quan trực tiếp thì để trống nội dung và trả FACTS là [].

{TIME_RULES}
{NEUTRAL_RULES}
{FORMAT_RULES}
- Mọi con số, ngày tháng, tên riêng phải đến từ dữ liệu mới. Cuối mỗi đoạn có dữ kiện ghi nguồn [S1], [S2].

DỮ LIỆU MỚI:
{ctx}

ĐỊNH DẠNG TRẢ VỀ (bắt buộc, không thêm gì khác; mỗi dòng ### đứng riêng một dòng):
###SECTION|{p['id']}
<nội dung bổ sung>
###FACTS|{p['id']}
<mảng JSON tối đa 8 dữ kiện, mỗi phần tử có các khóa: fact, numbers_dates (tối đa 5 mục, không liệt kê dải năm), year, type, sources, evidence, conflict>
(JSON phải hợp lệ: không dùng dấu ngoặc kép " bên trong giá trị chuỗi, không xuống dòng.)"""

	def run_expand(job):
		p, ctx, need = job
		try:
			raw = llm_call(expand_prompt(p, parsed[p["id"]]["text"], ctx, need), label="expand")
			return parse_batch(raw).get(p["id"])
		except Exception as e:
			return {"error": f"{type(e).__name__}: {str(e)[:150]}"}

	for _round in range(expand_rounds):
		expand_jobs = []
		for p in items:
			if p["kind"] != "section":
				continue
			r = parsed.get(p["id"])
			if not r or not r["text"]:
				continue
			have = word_count(r["text"])
			if have >= expand_min_ratio * p["target"]:
				continue
			ctx = get_section_context(index, p["query"], k=k_ctx, pool=pool_ctx, max_chars=ctx_chars, skip_seen=True)
			if len(ctx) < 1500:
				continue
			expand_jobs.append((p, ctx, max(100, p["target"] - have)))
		if not expand_jobs:
			break
		st.write(f"EXPAND vòng {_round + 1}: {len(expand_jobs)} section ngắn, viết tiếp bằng chunk mới")
		with ThreadPoolExecutor(max_workers=max(1, min(max_workers, len(expand_jobs)))) as ex:
			outs = list(ex.map(run_expand, expand_jobs))
		for (p, ctx, need), add in zip(expand_jobs, outs):
			if add and add.get("error"):
				st.write(f"  ⚠ expand lỗi [{p['query']}]: {add['error']}")
				continue
			if not add or not add["text"] or len(add["facts"]) < min_facts:
				continue
			r = parsed[p["id"]]
			paras = [x for x in re.split(r"\n+", r["text"]) if x.strip()]
			paras = paras[:-1] + [add["text"]] + paras[-1:] if len(paras) >= 2 else paras + [add["text"]]
			r["text"] = "\n".join(paras)
			r["facts"] += add["facts"]
			p["context"] = p["context"] + "\n\n" + ctx

	# 7.5 Kiểm tra bằng code + gắn nhãn độ mới
	results, all_facts, verify_jobs = [], [], []
	for p in items:
		r = parsed.get(p["id"], {"text": "", "facts": []})
		text = r["text"]
		facts = tag_freshness(r["facts"])

		if p["kind"] == "section":
			if not text:
				st.write(f"  → [{p['query']}] LLM trả nội dung rỗng/hỏng, bỏ qua section này")
				skipped.append({"query": p["query"], "title": p["title"], "text": "", "words": 0,
								"removed_claims": [], "status": "insufficient_data", "reason": "empty_text"})
				continue
			if len(facts) < min_facts:
				if word_count(text) < 80:
					st.write(f"  → [{p['query']}] quá ít dữ kiện ({len(facts)}) và text ngắn, bỏ qua section này")
					skipped.append({"query": p["query"], "title": p["title"], "text": "", "words": 0,
									"removed_claims": [], "status": "insufficient_data",
									"reason": f"facts={len(facts)}"})
					continue
				st.write(f"  → [{p['query']}] ít dữ kiện ({len(facts)}) nhưng text {word_count(text)} từ: giữ lại và ép verify")
				issues = ["no_facts"] + find_unsupported_numbers(text)
			else:
				issues, facts = check_section(text, facts)
			all_facts += facts
		else:
			if not text:
				st.write("  → Section lời khuyên trống, bỏ qua")
				continue
			issues = find_unsupported_numbers(text)
			all_facts += facts   # [SỬA4] facts của section lời khuyên cũng được tính vào has_current

		counts = {k: sum(1 for f in facts if f["freshness"] == k)
				  for k in ("current", "historical", "evergreen", "unknown")}
		entry = {"query": p["query"], "title": p["title"], "text": text,
				 "words": word_count(text), "removed_claims": [],
				 "verified": True, "verified_by": "code", "code_issues": issues,
				 "freshness_counts": counts, "status": "ok"}
		results.append(entry)
		if issues:
			verify_jobs.append((entry, p))
		st.write(f"  → [{p['query']}] {entry['words']} từ, nghi vấn từ code: {len(issues)}, dữ kiện: {counts}")

	# 7.6 LLM verify CHỈ cho section bị đánh dấu
	has_current = any(f["freshness"] == "current" for f in all_facts)
	if verify_jobs:
		st.write(f"VERIFY (LLM): {len(verify_jobs)}/{len(results)} section bị đánh dấu")
		order = {"current": 0, "evergreen": 1, "historical": 2, "unknown": 3}
		top_facts = sorted(all_facts, key=lambda f: order[f["freshness"]])[:20]

		def run_verify(job):
			entry, p = job
			try:
				if p["kind"] == "advice":
					return verify_section(entry["text"], json.dumps(top_facts, ensure_ascii=False, indent=1), strict=False)
				return verify_section(entry["text"], p["context"])
			except Exception as e:
				return entry["text"], [f"verify_error: {type(e).__name__}: {str(e)[:150]}"], False

		with ThreadPoolExecutor(max_workers=max(1, min(max_workers, len(verify_jobs)))) as ex:
			verify_out = list(ex.map(run_verify, verify_jobs))
		for (entry, p), (new_text, bad_claims, ok) in zip(verify_jobs, verify_out):
			# chặn verify cắt quá tay, hoặc trả về text hỏng
			if word_count(new_text) < 0.6 * max(entry["words"], 1) or is_degenerate(new_text):
				st.write(f"  ⚠ [{p['query']}] verify cắt quá nhiều/hỏng ({entry['words']} -> {word_count(new_text)} từ), giữ bản gốc")
				new_text = entry["text"]
				bad_claims = list(bad_claims) + ["verify_rejected"]
				ok = False
			entry["text"] = new_text
			entry["words"] = word_count(new_text)
			entry["removed_claims"] = bad_claims
			entry["verified"] = ok
			entry["verified_by"] = "llm"
			st.write(f"  → [{p['query']}] sau verify: {entry['words']} từ, loại {len(bad_claims)} câu, verified={ok}")

	warnings = []
	if not has_current:
		warnings.append(f"Không có dữ kiện nào từ năm {recent_from} trở lại đây. Bài sẽ thiếu số liệu hiện tại.")
		st.write("⚠ " + warnings[-1])

	# 7.7 Ghép bài
	results += skipped
	ok_sections = [r for r in results if r["status"] == "ok"]

	# loại câu trùng nguyên văn giữa các section
	seen_sent = set()
	for r in ok_sections:
		kept = []
		for para in r["text"].split("\n"):
			out_s = []
			for s in re.split(r"(?<=[.!?])\s+", para):
				key = norm(re.sub(r"\[S[^\]]*\]", "", s))
				if len(key) >= 40 and key in seen_sent:
					continue
				seen_sent.add(key)
				out_s.append(s)
			if out_s:
				kept.append(" ".join(out_s))
		r["text"] = "\n".join(kept)
		r["words"] = word_count(r["text"])

	article_tts = " ".join(clean_for_tts(r["text"]) for r in ok_sections)
	# [SỬA4] chốt chặn cuối: nếu bài cuối vẫn còn rác thì cảnh báo
	if is_degenerate(article_tts):
		warnings.append("Bài cuối vẫn có dấu hiệu rác (lặp số/JSON). Cần kiểm tra thủ công.")
		st.write("⚠ " + warnings[-1])
	sources = {sid: url for url, sid in global_source_ids.items()}
	unverified = [r["title"] for r in ok_sections if not r.get("verified")]
	if unverified:
		warnings.append(f"Section chưa được fact-check: {unverified}")
		st.write(warnings[-1])

	article_words = word_count(article_tts)
	success = len(ok_sections) > 0 and article_words >= min_total_words
	error = "" if success else (
		f"Chỉ có {len(ok_sections)} section hợp lệ, {article_words} từ (tối thiểu {min_total_words})."
	)
	if error:
		warnings.append(error)
	if target_total_words and article_words < 0.7 * target_total_words:
		warnings.append(f"Chỉ đạt {article_words}/{target_total_words} từ (nhiều section bị loại hoặc quá ngắn do thiếu dữ kiện).")
		st.write("⚠ " + warnings[-1])

	st.write(f"> DONE: {len(ok_sections)}/{len(results)} section, {article_words} từ, success={success}")
	return {
		"success": success,
		"error": error,
		"sections": results,
		"article_tts": article_tts,
		"sources": sources,
		"keywords": list_keywords,
		"has_current_data": has_current,
		"warnings": warnings,
	}
